-- 1. Crear un tipo ENUM para los roles y estados
CREATE TYPE user_role AS ENUM ('cliente', 'vendedor');
CREATE TYPE order_status AS ENUM ('pendiente', 'enviado', 'entregado', 'cancelado');

-- 2. Tabla de Perfiles (Extensión de la tabla auth.users de Supabase)
CREATE TABLE perfiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  email TEXT UNIQUE,
  nombre_completo TEXT,
  rol user_role DEFAULT 'cliente',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW())
);

-- 3. Tabla de Categorías
CREATE TABLE categorias (
  id SERIAL PRIMARY KEY,
  nombre TEXT NOT NULL UNIQUE
);

-- 4. Tabla de Productos (CRUD para Vendedores)
CREATE TABLE productos (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  vendedor_id UUID REFERENCES perfiles(id),
  nombre TEXT NOT NULL,
  descripcion TEXT,
  precio NUMERIC(10, 2) NOT NULL,
  imagen_url TEXT,
  stock INTEGER DEFAULT 0,
  categoria_id INTEGER REFERENCES categorias(id),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW())
);

-- 5. Tabla de Pedidos (Historial de compras)
CREATE TABLE pedidos (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente_id UUID REFERENCES perfiles(id),
  estado order_status DEFAULT 'pendiente',
  total NUMERIC(10, 2) DEFAULT 0,
  fecha TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW())
);

-- 6. Detalle de los Pedidos (Relación muchos a muchos entre productos y pedidos)
CREATE TABLE detalles_pedido (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  pedido_id UUID REFERENCES pedidos(id) ON DELETE CASCADE,
  producto_id UUID REFERENCES productos(id),
  cantidad INTEGER NOT NULL,
  precio_unitario NUMERIC(10, 2) NOT NULL
);

-- 8. Tabla de Carrito (para persistencia entre dispositivos)
CREATE TABLE carrito (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  usuario_id UUID REFERENCES perfiles(id) ON DELETE CASCADE,
  producto_id UUID REFERENCES productos(id) ON DELETE CASCADE,
  cantidad INTEGER DEFAULT 1,
  UNIQUE(usuario_id, producto_id)
);

-- 9. Funcion para mantener proteger el stock vs compras simpultaneas
CREATE OR REPLACE FUNCTION realizar_compra(
    p_producto_id UUID,
    p_cantidad INTEGER
)
RETURNS BOOLEAN
LANGUAGE plpgsql
AS $$
DECLARE
    v_stock INTEGER;
BEGIN

    -- Bloquear fila del producto
    SELECT stock
    INTO v_stock
    FROM productos
    WHERE id = p_producto_id
    FOR UPDATE;

    -- Verificar stock
    IF v_stock < p_cantidad THEN
        RETURN FALSE;
    END IF;

    -- Descontar stock
    UPDATE productos
    SET stock = stock - p_cantidad
    WHERE id = p_producto_id;

    RETURN TRUE;
END;
$$;

-- 10. Función para procesar el pedido completo de forma atómica (Calidad de Software)
CREATE OR REPLACE FUNCTION procesar_pedido_v2(p_usuario_id UUID)
RETURNS UUID
LANGUAGE plpgsql
AS $$
DECLARE
    v_pedido_id UUID;
    v_total NUMERIC(10, 2) := 0;
    v_item RECORD;
BEGIN
    -- 1. Calcular el total y verificar stock (con bloqueo de filas)
    -- Usamos un cursor para iterar sobre el carrito del usuario
    FOR v_item IN 
        SELECT c.producto_id, c.cantidad, p.precio, p.nombre, p.stock
        FROM carrito c
        JOIN productos p ON c.producto_id = p.id
        WHERE c.usuario_id = p_usuario_id
        FOR UPDATE OF p -- Bloquea las filas de productos para evitar cambios concurrentes
    LOOP
        -- Verificar stock
        IF v_item.stock < v_item.cantidad THEN
            RAISE EXCEPTION 'Stock insuficiente para el producto: % (Disponible: %, Solicitado: %)', 
                v_item.nombre, v_item.stock, v_item.cantidad;
        END IF;

        -- Sumar al total
        v_total := v_total + (v_item.precio * v_item.cantidad);
    END LOOP;

    -- Si el carrito está vacío, lanzamos error
    IF v_total = 0 THEN
        RAISE EXCEPTION 'El carrito está vacío o no existe.';
    END IF;

    -- 2. Crear la cabecera del pedido
    INSERT INTO pedidos (cliente_id, total, estado)
    VALUES (p_usuario_id, v_total, 'pendiente')
    RETURNING id INTO v_pedido_id;

    -- 3. Insertar detalles, descontar stock y limpiar carrito
    FOR v_item IN 
        SELECT producto_id, cantidad, precio
        FROM carrito 
        JOIN productos ON carrito.producto_id = productos.id
        WHERE usuario_id = p_usuario_id
    LOOP
        -- Descontar stock
        UPDATE productos 
        SET stock = stock - v_item.cantidad
        WHERE id = v_item.producto_id;

        -- Insertar detalle
        INSERT INTO detalles_pedido (pedido_id, producto_id, cantidad, precio_unitario)
        VALUES (v_pedido_id, v_item.producto_id, v_item.cantidad, v_item.precio);
    END LOOP;

    -- 4. Limpiar el carrito del usuario
    DELETE FROM carrito WHERE usuario_id = p_usuario_id;

    RETURN v_pedido_id;
END;
$$;

-- 11. Politicas para: Clientes solo ven sus pedidos, y Vendedores solo editan sus productos

CREATE POLICY "Usuarios ven sus pedidos"
ON pedidos
FOR SELECT
USING (auth.uid() = cliente_id);

CREATE POLICY "Vendedor edita sus productos"
ON productos
FOR UPDATE
USING (auth.uid() = vendedor_id);
